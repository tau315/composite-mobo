\section{Setup and notation}

Each benchmark is a deterministic multi-objective minimization problem with
a \emph{composite} structure
\[
    x \;\longrightarrow\; g(x)\in\R^{K} \;\longrightarrow\; f(x)=L\big(g(x)\big)\in\R^{M},
\]
where $g$ is a \emph{raw response} that a real simulator/solver computes and
$L$ is a \emph{known}, closed-form reduction. The \textbf{composite} variant of
a solver fits $K$ surrogates to $g$ and applies $L$ analytically; the
\textbf{direct} variant fits $M$ surrogates to $f$. Inputs are normalized to
$[0,1]^d$ and mapped to each problem's native bounds.

\paragraph{Solvers.} \texttt{standard\_mobo} (qEHVI), \texttt{chebyshev\_bo}
(augmented-Chebyshev scalarization), \texttt{spherical\_chebyshev\_bo}
(spherical scalarization, high-dim), \texttt{batched\_morbo} (trust-region
MORBO); each in a direct and composite variant.
Reported $\Delta$ is the composite-minus-direct percentage change in final
dominated hypervolume, paired per seed at $n$ seeds ($*$: $p<0.05$, paired
$t$-test).

\section{Synthetic composite benchmarks}

\subsection{DTLZ2 ($M{=}2$; $d\in\{6,100,600\}$)}
With one position variable $x_0$ and distance variables $x_1,\dots,x_{d-1}$,
\[
g = \sum_{j=1}^{d-1}(x_j-0.5)^2,\qquad
\theta=\tfrac{\pi}{2}x_0,\qquad
f_1=(1+g)\cos\theta,\ \ f_2=(1+g)\sin\theta .
\]
\textbf{Composite:} $g(x)$ exposes the per-variable terms $(x_j-0.5)^2$
(each a $1$-D function); $L$ sums them to $g$ and applies the trig map.
\textbf{Result} (\texttt{batched\_morbo}): $d{=}6$ \win{$+5.3\%$*}, $d{=}100$
\win{$+0.3\%$*}, $d{=}600$ $+0.1\%$. Also $M{=}5$, $d{=}6$: \win{$+0.5\%$*}.

\subsection{Ackley--Griewank ($M{=}2$; $d\in\{6,50\}$)}
On rotated/shifted coordinates $z=R(x-c)$, the two objectives are a
normalized Ackley and Griewank. Ackley uses the two intermediates
$\big(\overline{z^2},\ \overline{\cos 2\pi z}\big)$ with
\[
\mathrm{Ackley}= -20\exp\!\big(-0.2\sqrt{\overline{z^2}}\big)-\exp\!\big(\overline{\cos 2\pi z}\big)+20+e ;
\]
Griewank uses $\big(\tfrac{1}{4000}\sum z_j^2,\ \prod_j\cos(z_j/\sqrt{j})\big)$
with $\mathrm{Griewank}=1+\tfrac{1}{4000}\sum z_j^2-\prod_j\cos(z_j/\sqrt{j})$.
\textbf{Composite:} $g(x)$ exposes these mean/sum/product intermediates; $L$
is the outer nonlinearity. \textbf{Result:} $d{=}6$ \win{$+2.7\%$*}
(\texttt{standard}); $d{=}50$ $+0.1\%$ (\texttt{batched\_morbo}).

\subsection{Five-Ackley ($M{=}5$, $d{=}6$) and Langermann--Ackley ($M{=}2$, $d{=}6$)}
Multi-objective stacks of the Ackley intermediates above and of the
generalized Langermann map
$-\sum_i c_i \exp(-d_i/\pi)\cos(\pi d_i)$ (with $d_i$ squared distances to
targets). \textbf{Composite:} the per-objective additive terms are the raw
response. \textbf{Result} (\texttt{batched\_morbo}): Five-Ackley
\win{$+5.0\%$*}; Langermann--Ackley \win{$+5.1\%$*} (and \win{$+8.0\%$*} on
\texttt{standard}).

\subsection{Projected Langermann ($M{=}2$, $d{=}500$)}
A $500$-D Langermann whose raw response exposes the low-dimensional
projections feeding the exponential-cosine map. \textbf{Result}
(\texttt{spherical\_chebyshev}): \win{$+0.5\%$*}.

\section{Real-world benchmarks}

\subsection{Summit SnAr plug-flow reactor ($M{=}2$, $d{=}4$)}
Inputs: residence time $\tau$, pyrrolidine equivalents $\rho$, inlet
concentration $C_0$, temperature $T$. The five species
$C=(C_{\text{dfnb}},C_{\text{pldn}},C_{\text{prod}},C_{\text{regio}},C_{\text{bis}})$
evolve under Arrhenius kinetics
$k_j=0.6\,k_j^{\text{ref}}\exp[-\tfrac{E_{a,j}}{R}(\tfrac{1}{T_K}-\tfrac{1}{T_{\text{ref}}})]$:
\[
\begin{aligned}
\dot C_0 &= -(k_a{+}k_b)C_0C_1, &
\dot C_2 &= k_aC_0C_1-k_cC_1C_2,\\
\dot C_1 &= -(k_a{+}k_b)C_0C_1-k_cC_1C_2-k_dC_1C_3, &
\dot C_3 &= k_aC_0C_1-k_dC_1C_3,\\
\dot C_4 &= k_cC_1C_2+k_dC_1C_3, & &
\end{aligned}
\]
integrated over $[0,\tau]$ with flow $q=V/\tau$ ($V{=}5$ mL). Objectives:
\[
\mathrm{STY}=\tfrac{6\times10^4}{1000}\tfrac{M_2\,C_{\text{prod}}\,q}{V}\ (\text{max}),\qquad
\mathrm{E}=\frac{q\rho_{\text{EtOH}}+10^{-3}q\sum_{i\neq2}M_iC_i}{10^{-3}M_2C_{\text{prod}}q}\ (\text{min}).
\]
\textbf{Composite:} $g(x)=(C,\,F_{\text{prod}})$ with product molar flow
$F_{\text{prod}}=C_{\text{prod}}q$; STY reads $F_{\text{prod}}$ and $q$ cancels
in E (leaving a function of $C$ only). \textbf{Result:} \win{$+0.3\%$*}
(\texttt{batched\_morbo}) --- essentially tied (a coupled 4-D reactor ODE).

\subsection{Nanoparticle RGB structural color ($M{=}3$, $d{=}6$)}
Inputs: core radius and five shell thicknesses. Mie scattering is evaluated at
$201$ visible wavelengths, giving a scattering spectrum $S(\lambda;x)$. Each
color objective $m\in\{R,G,B\}$ is composed from its in-band and out-of-band
integrated intensities,
\[
I^{\text{in}}_m=\!\!\int_{\text{band }m}\!\!\!S\,d\lambda,\quad
I^{\text{out}}_m=\!\!\int_{\text{out }m}\!\!\!S\,d\lambda,\qquad
f_m = L\big(I^{\text{in}}_m,I^{\text{out}}_m\big).
\]
\textbf{Composite:} $g(x)=(I^{\text{in}}_m,I^{\text{out}}_m)_{m}$ (the band
integrals); $L$ forms the selectivity objectives. \textbf{Result:} $+0.2\%$
(\texttt{standard}) --- tied (a coupled global Mie solve).

\subsection{CORT TG119 radiotherapy ($M{=}3$, $d{=}418$)}
$418$ beamlet intensities $x\ge0$; public dose-influence matrices $D_s$ per
structure $s$ give the voxel dose $\mathbf{d}_s=D_s x$. Objectives are
target-coverage / organ-sparing functionals of the per-voxel doses.
\textbf{Composite:} $g(x)$ exposes the per-structure dose statistics; $L$ forms
the three clinical objectives. \textbf{Result:} \win{$+1.5\%$*}
(\texttt{spherical\_chebyshev}) but \loss{$-2.0\%$*} (\texttt{batched\_morbo})
--- a sign flip by solver (method dependence).

\subsection{Penicillin fermentation ($M{=}3$, $d{=}7$)}
A fed-batch bioreactor ODE integrated over the run; the raw response is the
checkpointed state trajectory $g(x)=\{s(x,t_k)\}_k$, reduced to
$(\text{yield},\,\mathrm{CO_2},\,\text{time})$. \textbf{Result:}
\loss{$-2.5\%$*} (\texttt{batched\_morbo}) --- a coupled trajectory of one
process; composite hurts.

\section{Power-flow benchmarks (MOBO-HighDim contribution)}

Real IEEE 14-bus optimal power flow (CEC2021 RWCMOP). With complex bus
voltages $V$ (design variables, slack $V_1{=}1$) and sparse admittance
$Y=G+jB$, one solve gives the per-bus complex power
\[
I = Y V,\qquad S = V\odot\bar I,\qquad P_{sp}=\Re(S),\ \ Q_{sp}=\Im(S).
\]
Because $Y$ is \emph{sparse}, each $P_{sp,k}$ depends only on bus $k$'s
neighbors --- the source of decoupling. The raw response is
$g(x)=(P_{sp},Q_{sp},\dots)$ and objectives are decoupled sums over it.

\subsection{RCM40 ($M{=}2$, $d{=}34$)}
$f_1=\sum_k P_{sp,k}$ (active loss), $f_2=-\Im(I_2)+\sum_{k\ge2}Q_{sp,k}$
(reactive loss). \textbf{Result:} \win{$+0.8\%$*} (\texttt{batched\_morbo}).

\subsection{RCM46 ($M{=}4$, $d{=}34$)}
Adds fuel cost $f_1=\sum_k(b_kP_{g,k}+c_kP_{g,k}^2)$ and voltage deviation
$f_4=\sum_{k\ge2}(1-|V_k|)^2$ to RCM40's two loss objectives.
\textbf{Result:} \win{$+5.4\%$*} (\texttt{batched\_morbo}), \win{$+19.0\%$*}
(\texttt{spherical\_chebyshev}). The win grows from RCM40's $2$ to RCM46's
$4$ decoupled objectives.

\subsection{MOOPF-5 ($M{=}5$, $d{=}34$; constructed)}
RCM46's four objectives plus the Kessel--Glavitsch L-index voltage-stability
margin. Partitioning $Y$ into generator (PV/slack) and load blocks and forming
$F=-Y_{LL}^{-1}Y_{LG}$,
\[
L_j=\Big|\,1-\sum_{i\in\text{gen}}F_{ji}\,\tfrac{V_i}{V_j}\,\Big|,\qquad f_5=\max_{j\in\text{load}}L_j .
\]
$g(x)$ appends the per-load-bus $L_j$ to RCM46's response.
\textbf{Result:} \win{$+23.5\%$*} (\texttt{batched\_morbo}, $n{=}20$),
\win{$+14.6\%$*} (\texttt{spherical\_chebyshev}) --- the largest win in the
suite. (Not a CEC2021 case; a constructed benchmark to test whether more
decoupled objectives amplify the advantage.)

\section{Controlled decoupling benchmark}

\subsection{Blocked DTLZ2 ($M{=}2$, $d{=}49$)}
DTLZ2 with the objective held fixed while the distance term is exposed as
$K=\lceil (d{-}1)/b\rceil$ block partial sums,
\[
\text{block}_i=\sum_{j\in\text{block }i}(x_j-0.5)^2,\qquad \sum_i\text{block}_i=g\ \ \forall b .
\]
The block size $b$ dials the effective input-dimension of each raw component
($b/d$) from $\sim1/d$ (decoupled) to $1$ (coupled) with everything else
fixed. \textbf{Result:} composite advantage falls monotonically from
$+11.9\%$ ($b{=}1$) to $+7.4\%$ ($b{=}48$), Pearson $r=-0.997$
(\texttt{ablation\_results/\_decoupling\_sweep/}). Causal confirmation that
decoupling drives the advantage.

\section{Key findings}
\begin{enumerate}
  \item \textbf{Wins scale with decoupled-objective count} on the identical
  IEEE 14-bus system: RCM40 ($+0.8\%$) $\to$ RCM46 ($+5.4\%$) $\to$ MOOPF-5
  ($+23.5\%$).
  \item \textbf{Method dependence:} CORT flips sign between
  \texttt{batched\_morbo} ($-2.0\%$) and \texttt{spherical\_chebyshev}
  ($+1.5\%$); the advantage is strongest with trust-region acquisition.
  \item \textbf{Chemistry / trajectory / global-solve benchmarks do not
  benefit} (SnAr, Penicillin, CORT-on-morbo): the raw response is one
  globally-coupled state, no easier to model than the objective.
  \item \textbf{Decoupling, not reduction nonlinearity, is the discriminator:}
  RCM46 wins with a linear reduction while SnAr's nonlinear E-factor ratio
  ties. The effective input-dimension of $g$ separates wins from losses.
  \item \textbf{Causal confirmation (blocked DTLZ2):} varying only the
  raw-response effective dimension moves the advantage monotonically,
  $r=-0.997$.
\end{enumerate}
